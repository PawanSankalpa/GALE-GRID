## ⚡ Quick Reference

### The One-Minute Version
```jsx
// 1. Import
import { SkeletonGrid } from '../components/skeletons/SkeletonComponents';
import { useSkeletonDelay } from '../hooks/useDataLoader';

// 2. Add state
const [data, setData] = useState(null);
const [loading, setLoading] = useState(true);

// 3. Load data
useEffect(() => {
  fetch('/api/data').then(r => r.json()).then(d => {
    setData(d);
    setLoading(false);
  });
}, []);

// 4. Use hook
const showSkeleton = useSkeletonDelay(loading);

// 5. Render
return showSkeleton ? <SkeletonGrid /> : <div className="fade-in">{content}</div>;
```

### The Most Common Components
- **Grid of items:** `<SkeletonGrid count={3} />`
- **Carousel:** `<SkeletonCarousel count={3} height="250px" />`
- **Cards:** `<SkeletonCard showAvatar={true} lines={3} />`
- **Text:** `<SkeletonText lines={3} />`
- **Images:** `<SkeletonImage aspect="landscape" />`

### The Most Important Hook
```jsx
const showSkeleton = useSkeletonDelay(loading, 300);
// Shows skeleton only if loading > 300ms
// Prevents flashing on fast networks
```

## 📊 What Changes in Your Codebase

### Before: PortfolioSection.jsx (Current)
```jsx
function PortfolioSection() {
  // No loading state
  const [portfolio] = useState(PORTFOLIO_DATA);
  
  return <div>{portfolio.map(...)}</div>;
}
```

### After: Using Skeleton System
```jsx
function PortfolioSection() {
  // Add loading state
  const [portfolio, setPortfolio] = useState(null);
  const [loading, setLoading] = useState(true);
  
  // Load data
  useEffect(() => {
    loadData().then(d => {
      setPortfolio(d);
      setLoading(false);
    });
  }, []);
  
  // Use hook
  const showSkeleton = useSkeletonDelay(loading);
  
  // Conditional render
  return showSkeleton ? 
    <SkeletonPortfolioGrid /> : 
    <div className="fade-in">{portfolio.map(...)}</div>;
}
```
